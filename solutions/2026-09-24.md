给定一个长度为 $n$ 的序列，划分为 $k$ 组，初始未知划分方案
每次允许询问一个集合，交互器返回询问集合内被完全包含的分组数量，要求不超过 $n \lceil \log n\rceil$ 次询问得出初始划分。

交互次数与 $\texttt{log}$ 相关，优先考虑二分和二进制。

核心观察:

- 对于一个未知的集合，一定存在一个最左端点和最右端点，并且按端点询问必定具备单调性。

- 考虑每次对未知下标集合进行询问，每次二分出一个最右端点，使得交互器返回恰好包含一个集合。这样询问出的最右端点一定是这个集合内部的。

-  这样二分每个数的询问次数严格不超过 $\lceil \log n \rceil$ 次

```cpp
#include<bits/stdc++.h>
using namespace std;
//#define endl '\n'交互需注释
typedef long long ll;
void solve(){
    int n,k;
    cin>>n>>k;
    int cnt=n/k;
    auto ask=[&](const vector<int>&S) ->int {
        cout<<"? "<<S.size()<<' ';
        for(int v:S) cout<<v<<' ';
        cout<<endl;
        int ans;cin>>ans;
        return ans;
    };
    vector<int>ans(n+1);
    vector<int>used(n+1);
    for(int j=1;j<cnt;j++){
        vector<int>S;
        vector<int>cur;
        for(int i=1;i<=n;i++) if(!used[i]) S.emplace_back(i);
        for(int t=1;t<=k;t++){
            int l=0,r=S.size();
            while(l<=r){
                int mid=l+r>>1;
                vector<int>T=cur;
                for(int idx=0;idx<mid;idx++) T.emplace_back(S[idx]);
                if(ask(T)>0) r=mid-1;
                else l=mid+1;
            }
            int val=S[l-1];
            used[val]=1;
            ans[val]=j;
            cur.emplace_back(val);
            S.erase(S.begin()+l-1);
        }
    }//对n/k个集合进行询问,平均每个数询问logn次
    for(int i=1;i<=n;i++) if(!ans[i]) ans[i]=cnt;
    cout<<"! ";
    for(int i=1;i<=n;i++) cout<<ans[i]<<' ';
    cout<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    solve();
    return 0;
}
```
