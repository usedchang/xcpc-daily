可以把题目看作一个`背包`类的问题，但是有一个条件十分麻烦`当多件装备同时触发暴击时，只有最高的倍率会生效。`。在这个条件下每次枚举一个装备选择后暴击有可能无法生效，思考如何去掉这个限制，比较直接的想法是增加一个背包维度重新设计状态，显然这样时空复杂度都无法通过。

如果我们对物品按倍率进行一个排序，那么就可以发现每次枚举物品，如果产生暴击就一定能够生效。
先定义 $\texttt{dp}$ 状态, $dp_{i,w}$ 表示前 $i$ 个物品耗费 $w$ 的代价下查收的最大倍率。
我们有：

$$
dp_{i,w}=\max(dp_{i-1,w},(1-\frac{p_i}{100})dp_{i-1,w-w_i}+\frac{p_i}{100}v_i)
$$

```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
typedef long double ld;
struct node{
    ll p,v,w;
    node(ll p,ll v,ll w):p(p),v(v),w(w){};
    node(){};
};
void solve(){
    int n,m;
    cin>>n>>m;
    vector<node>a(n+1);
    for(int i=1;i<=n;i++) cin>>a[i].p>>a[i].v>>a[i].w;
    sort(a.begin()+1,a.end(),[&](node &A,node &B){
        return A.v<B.v;
    });//保证严格偏序下效益一定能生效
    vector<ld>dp(m+1);
    for(int i=1;i<=n;i++){
        for(int j=m;j>=a[i].w;j--){
            dp[j]=max(dp[j],(1.0L-a[i].p/100.00L)*dp[j-a[i].w]+a[i].p/100.00L*a[i].v);
        }
    }
    cout<<dp[m]<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T;cin>>T;
    cout<<fixed<<setprecision(12);
    while(T--) solve();
    return 0;
}
```
