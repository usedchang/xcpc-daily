简单博弈题
**核心观察** :
- 对于多个堆数量相同时,可以视为一个堆，因为一定会对多个堆同时操作
- 把多个堆排序后，当前每一堆能够取的数量严格不超过最小堆的数量
- 当能够取的数量达到`2`以上时,一定可以取到必胜态:具体的,如果每次只能取`1`个时,答案只由奇偶性确定。如果能取两个以上一定拥有交换先后手的机会。因此对于，对于只有必胜与必败态的博弈而言，第一次拥有交换先后手的机会时，一定能取到必胜态。

按照以上观察可以得出代码

```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll inf=1e18;
struct node{
    int x,y,z;
};
void solve(){
    int n;cin>>n;
    vector<ll>a(n+1);
    for(int i=1;i<=n;i++) cin>>a[i];
    sort(a.begin()+1,a.end());
    a.erase(unique(a.begin()+1,a.end()),a.end());
    n=a.size()-1;
    int op=1;
    for(int i=1;i<=n;i++){
        if(a[i]-a[i-1]!=1){
            if(op){
                cout<<"Alice"<<endl;
                return;
            }
            else{
                cout<<"Bob"<<endl;
                return;
            }
        }
        op^=1;
    }
    if(op) cout<<"Bob"<<endl;
    else cout<<"Alice"<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T=1;cin>>T;
    while(T--) solve();
    return 0;
}
```
