先考虑暴力应该怎么写，对于给定的 $n$ 和 $m$ 暴力计算可以枚举 $n$ 取 [1,n]时最小累加m的代价取min，对于枚举的代价是$(n'-m \bmod n') \bmod n'+(n-n')$。

但这样算显然会高达 $O(n)$ 的复杂度，有没有什么更为高效的方法呢？

有的，设 $m=xn+b$,发现一定满足$x \leq \sqrt m$ 或 $n \leq \sqrt m$,枚举两者中作为较小值的情况分别计算可以得出答案,又因为 $m$的上界仅有 $10^8$,$O(T \sqrt{m})$的复杂度足以通过此题。
```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll INF=1e18;
void solve(){
    ll n,m;
    cin>>n>>m;
    ll ans=INF,cnt=0;
    for(int x=1;x<=10000;x++){
        ll dn=(m+x-1)/x;
        if(dn<=n&&dn>0){
            ans=min(ans,(n-dn)+(dn-m%dn)%dn);
        }
    }
    for(int dn=1;dn<=10000;dn++){
        ll x=m/dn;
        if(dn<=n&&dn>0){
            ans=min(ans,(n-dn)+(dn-m%dn)%dn);
        }
    }
    cout<<ans<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T;cin>>T;
    while(T--) solve();
    return 0;
}
```

对于原式代价 $(n'-m \bmod n') \bmod n'+(n-n')$

可以看作 $n-n'+(\lceil \frac{m}{n'}\rceil \times n' -m)$
显然也可以根据恒等式 $\lceil \frac{m}{n}\rceil = \lfloor \frac{m-1}{n}\rfloor+1$ 使用`整除分块`这一算法解决这题。
```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll INF=1e18;
void solve(){
    ll n,m;
    cin>>n>>m;
    ll ans=INF;
    for(ll l=1,r;l<=n;l=r+1){
        ll val=(m-1)/l;
        if(val==0){
            ans=min(ans,n-m);
            break;
        }
        r=min(n,(m-1)/val);
        ans=min(ans,(n-m)+val*l);
    }
    cout<<ans<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T;cin>>T;
    while(T--) solve();
    return 0;
}
```
