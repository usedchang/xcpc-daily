这是一道经典的 `倍数容斥` 类题目，有两种角度思考，首先我们要考虑到直接枚举计算长为 $K$ 的二进制重复串比较困难，我们考虑计算不是二进制重复串的那些字符串，发现对于一个长为 $n$ 的二进制串， $f(n)=2^n - g(n)$ ,其中 $g(n)$表示那些不是二进制重复穿的字符串(另记作"本原串"),$f(n)$ 表示合法字符串的数量。
发现从小到大枚举可以消除本原串计数对较小数的依赖，具体的：
- 初始化 $g(n) =2^n$。
- 然后从小到大枚举 $g(n)$ 的倍数，不断消除枚举数的倍数对枚举数的依赖。
- 算到 $g(i)$ 时，保证 $g(j),j\mid i$ 的贡献一定已经被消除了。

方法一:从小到大消除依赖
```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll mod=1e9+7;
const int N=1e6+5;
ll ans[N],f[N],_2[N];
void init(int n){
    vector<ll>f(n+1,1);//本原串的数量
    for(int i=1;i<=n;i++) f[i]=f[i-1]*2%mod,_2[i]=f[i];
    for(int i=1;i<=n;i++){
        for(int j=2*i;j<=n;j+=i){
            f[j]-=f[i];
            if(f[j]<0) f[j]+=mod;
        }
        ans[i]=_2[i]-f[i];
        if(ans[i]<0) ans[i]+=mod;
    }
}
void solve(){
    ll n;cin>>n;
    cout<<ans[n]<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    init(1e6);
    int T=1;cin>>T;
    while(T--) solve();
    return 0;
}
```



方法二:莫比乌斯反演
对于倍数容斥类题目，我们都可以推式子转化莫比乌斯反演意义下做。

记 $f(n)为所有长为n的串的数量，g(n)表示长为n的本原串的数量(就是不可以被分割为更短重复串的那些字符串)$ 
则有$f(n)=2^n= \sum_{d \mid n} g(d)$，这个式子的意义就是每个二进制字符串都有最小的周期。根据莫比乌斯反演公式，有 $g(n)=\sum_{d \mid n} 2^{n/d} \mu(d)$

所以答案就是$2^n-g(n)=2^n-\sum_{d \mid n} 2^{n/d} \mu(d)$.
```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const int N=1e6+5;
int mu[N];
bool isp[N];
ll pw[N],ans[N];
const ll mod=1e9+7;
vector<int>p;
void init(int n){
    pw[0]=1;
    mu[1]=1;
    isp[1]=false;
    memset(isp,true,sizeof isp);
    for(int i=1;i<=n;i++) pw[i]=pw[i-1]*2%mod;
    for(int i=2;i<=n;i++){
        if(isp[i]){
            p.push_back(i);
            mu[i]=-1;
        }
        for(int x:p){
            if(x*i>n) break;
            isp[i*x]=false;
            mu[i*x]=-mu[i];
            if(i%x==0) {
                mu[i*x]=0;
                break;
            }
        }
    }
    for(int i=1;i<=n;i++){
        ans[i]+=pw[i];
        for(int j=i;j<=n;j+=i){
            ans[j]-=mu[i]*pw[j/i]%mod;
            ans[j]+=mod;
            ans[j]%=mod;
        }   
    }
}
void solve(){
    /*
    f_n=\sum_{d \mid n} G(n)
    G_n= \sum_{d \mid n} mu(d) \times f_n
    G_n= \sum_{d \mid n} mu(d) 2^(n/d) 
    2^n-mu(d) 2^(n/d)
    */
    ll n;cin>>n;
    cout<<ans[n]<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    init(1e6);
    int T=1;cin>>T;
    while(T--) solve();
    return 0;
}
```
