核心观察:

- 对于路径权值按位与之并大于等于 $V$ 的可以按位考虑，就是与 $V$在高位上相等，某一位上恰好超越或者所有位全部相等，最多一共只有 $60$种不同情况。

- 是否存在路径可以看成是否联通，转化成连通性的判定问题，考虑使用并查集。

复杂度$O(n \log{V} \log{\log{V}})$，如果按秩合并可以做到更快，但这题并不卡常，普通并查集足以通过。


```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
struct DSU{
    vector<int>f;
    DSU(int n){
        f.resize(n+1);
        iota(f.begin(),f.end(),0);
    }
    int find(int x){
        return x==f[x]?x:f[x]=find(f[x]);
    }
    bool merge(int x,int y){
        int fx=find(x),fy=find(y);
        if(fx==fy) return false;
        f[fx]=fy;
        return true;
    }
    bool same(int x,int y){return find(x)==find(y);}
};
struct edge{
    int u,v;ll w;
    edge(){};
    edge(int u,int v,ll w):u(u),v(v),w(w){};
};
void solve(){
    int n,m,q;ll v;
    cin>>n>>m>>q>>v;
    vector<edge>E;
    vector<edge>nxt;
    set<pair<ll,ll>>st;
    vector<DSU>D(60,DSU(n+1));
    edge tmp;
    for(int i=1;i<=m;i++) {
        cin>>tmp.u>>tmp.v>>tmp.w;
        E.emplace_back(tmp);
    }
    for(int j=59;j>=0;j--){
        if(v>>j&1){
            for(edge &p:E){
                if(p.w>>j&1) nxt.emplace_back(p);
            }
            E=move(nxt);
        }
        else{
            for(edge &p:E){
                if(p.w>>j&1) D[j].merge(p.v,p.u);
            }
        }
    }
    DSU equ(n+1);
    for(edge &p:E) equ.merge(p.u,p.v);
    while(q--){
        int u,v;
        cin>>u>>v;
        if(equ.same(u,v)) cout<<"Yes"<<endl;
        else {
            bool f=false;
            for(int j=59;j>=0;j--){
                if(D[j].same(u,v)) {
                    f=true;
                    break;
                }
            }
            if(f) cout<<"Yes"<<endl;
            else cout<<"No"<<endl;
        }
    }
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T=1;//cin>>T;
    while(T--) solve();
    return 0;
}
```
