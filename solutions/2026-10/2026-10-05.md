核心观察:

- 取每次生效的值一定构成一段单调不减的序列,且只有递增或末尾的时候才需要考虑。

- 每次生效的值上界不超过 $k$ ,本质需要转移的位置最多 $k+1$ 个

- 可以用一个二维的 $dp_{s,m}$ 表示在取到和为 $s$ 最大值为 $m$ 时的总收益

根据以上可以给出大致的 $dp$ 方程

$dp_{s,m}=\max(dp_{s,m},pre_{s-i,m} + {i} \times {len})$

$dp_{s,m}=\max(dp_{s,m},pre_{s,m} + {m} \times {len})$

这样子可以写出一个$O(k^4+n)$ 的TLE的代码
```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll INF=1e18;
void solve(){
    int n,k;
    cin>>n>>k;
    vector<ll>a(n+1);
    vector<pair<ll,ll>>seg;
    int las=0,len=0,mx=0;
    for(int i=1;i<=n;i++) {
        cin>>a[i];
        if(mx<a[i]){
            mx=a[i];
            len=i-las;
            seg.emplace_back(mx,len);
            las=i;
        }
    }
    seg.emplace_back(mx,n-las+1);
    vector<vector<ll>>dp(k+1,vector<ll>(k+1,-INF));//代价和为i,代价最大值为j的时候总效益
    dp[0][0]=0;
    for(auto &[mx,len]:seg){
        vector<vector<ll>>nxt=dp;
        for(int m=0;m<=mx;m++){
            for(int s=m;s<=k;s++){
                for(int i=0;i<=s;i++){
                    if(m>=i) nxt[s][m]=max(nxt[s][m],dp[s-m][i]+1LL*i*len);
                }
                nxt[s][m]=max(nxt[s][m],dp[s][m]+1LL*m*len);
            }
        }
        dp=move(nxt);
    }
    ll ans=0;
    for(int s=0;s<=k;s++){
        for(int j=0;j<=s;j++) ans=max(ans,dp[s][j]);
    }
    cout<<ans<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T;cin>>T;
    while(T--) solve();
    return 0;
}
```

但是以上代码显然无法通过此题！
考虑优化，其中一段发现转移方式固定，可以对其使用前缀和优化可以做到 $O(k^3+n)$ 的复杂度。这个复杂度足以通过本题。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const ll INF=1e18;
void solve(){
    int n,k;
    cin>>n>>k;
    vector<ll>a(n+1);
    vector<pair<ll,ll>>seg;
    int las=0,len=0,mx=0;
    for(int i=1;i<=n;i++) {
        cin>>a[i];
        if(mx<a[i]){
            mx=a[i];
            len=i-las;
            seg.emplace_back(mx,len);
            las=i;
        }
    }
    seg.emplace_back(mx,n-las+1);
    vector<vector<ll>>dp(k+1,vector<ll>(k+1,-INF));//代价和为i,代价最大值为j的时候总效益
    dp[0][0]=0;
    for(auto &[mx,len]:seg){
        vector<vector<ll>>nxt=dp;
        vector<vector<ll>>pre(k+1,vector<ll>(k+1,-INF));
        for(ll s=0;s<=k;s++){
            pre[s][0]=0;
            for(int m=1;m<=mx;m++) pre[s][m]=max(pre[s][m-1],dp[s][m]+1LL*m*len);
        }
        for(int m=0;m<=mx;m++){
            for(int s=m;s<=k;s++){
                nxt[s][m]=max(nxt[s][m],pre[s-m][m]);
                nxt[s][m]=max(nxt[s][m],dp[s][m]+1LL*m*len);
            }
        }
        dp=move(nxt);
    }
    ll ans=0;
    for(int s=0;s<=k;s++){
        for(int j=0;j<=s;j++) ans=max(ans,dp[s][j]);
    }
    cout<<ans<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T;cin>>T;
    while(T--) solve();
    return 0;
}
```

bonus:本题可以使用斜率优化的手段做到 $O(k^2\log k+n)$
