简要题意:选择 $S$ 中一段子串，$T$ 中小于这段子串长度一段前缀串，然后求这样拼接而成的字符串是一个回文串的方案数。

核心观察:

- 回文中心一定出现在较长子串中 ,即 $S$ 的子串中

- 拼接而成的串回文，把 $T$ 的一段前缀记作$t$,而 $rt$ 表示翻转后的这段前缀。那么构成答案的回文串可以表示为 $rt+mid+t$。其中 $mid$ 一定满足是回文串。

这样我们需要预处理出与 $rt$ 的前缀匹配每个位置 $i$ 结尾的最长匹配长度 $las_i$，这里我们考虑使用 [Z函数](https://oi-wiki.org/string/z-func/)或者字符串哈希预处理解决。

然后我们需要通过[Manacher](https://oi-wiki.org/string/manacher/)来计算出 $S$ 中以每个位置 $i$ 开始的最长回文长度 $len_i$。

这样答案就是 $\sum_{i=1}^{n-1} {las_i}{len_{i+1}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
vector<int>zf(const string &s){
    int n=s.size();
    vector<int>z(n);
    z[0]=n;
    for(int i=1,l=0,r=0;i<n;i++){
        if(i<r) z[i]=min(r-i,z[i-l]);
        while(i+z[i]<n&&s[z[i]]==s[i+z[i]]) z[i]++;
        if(i+z[i]>r){
            l=i,r=i+z[i];
        }
    }
    return z;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    string s,t;
    cin>>s>>t;
    int n=s.size(),m=t.size();
    string rs=s;
    reverse(rs.begin(),rs.end());
    vector<int>z=zf(t+'#'+rs);
    vector<int>las(n+1);
    for(int j=0;j<n;j++) las[j]=z[m+n-j];
    string T="$#";
    for(char c:s) T+=c,T+='#';
    T+='^';
    int lenT=T.size();
    vector<int>p(lenT);
    for(int i=1,mx=0,mid=0;i<lenT;i++){
        if(i<mx) p[i]=min(mx-i,p[2*mid-i]);
        else p[i]=1;
        while(T[i-p[i]]==T[i+p[i]]) ++p[i];
        if(i+p[i]>mx){
            mx=i+p[i];
            mid=i;
        }
    }
    vector<ll>d(n+3);
    for(int i=1;i<lenT;i++){
        int len=p[i]-1;
        if(!len) continue;
        int l=(i-len-1)/2;
        int s=(len+1)/2;
        d[l]++;
        d[l+s]--;
    }
    for(int i=1;i<=n;i++) d[i]+=d[i-1];
    ll ans=0;
    for(int i=0;i<n;i++) ans+=las[i]*d[i+1];
    cout<<ans<<endl;
    return 0;
}
```
