好史的题意

简要题意:有一张动态有向图，边可被摧毁/修复，摧毁或修复据点 $u$ 等价于批量摧毁或修复所有指向 $u$ 的入边。每次操作后，如果当前每个点恰好只有一条可用出边，则输出 `YES`，否则输出 `NO`。

发现这其实转化成了一个内向基环树森林的判定性问题。

要如何判定是否是一个内向基环树呢？比较简单的做法是给每一个节点随机分配一个独立的权值，然后动态的维护每条出边指向点哈希值的和。

判定任意时刻所有不同权值哈希值的和是否与当前哈希值的和相等。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
typedef unsigned long long ull; 
const int N=5e5+5;
ull a[N];
ull pre[N],del[N];
int x,y,n,m,q;
void solve(){
    mt19937_64 rng(time(0));
    cin>>n>>m;
    ull to=0;
    ull sum=0;
    for(int i=1;i<=n;i++) a[i]=rng(),to+=a[i];
    for(int i=1;i<=m;i++){
        cin>>x>>y;
        pre[y]+=a[x]; 
        sum+=a[x];
    }
    cin>>q;
    while(q--){
        int op;cin>>op;
        if(op==1){
            cin>>x>>y;
            pre[y]-=a[x];
            del[y]+=a[x];
            sum-=a[x];
        }
        else if(op==2){
            cin>>y;
            sum-=pre[y];
            del[y]+=pre[y];
            pre[y]=0;
        }
        else if(op==3){
            cin>>x>>y;
            pre[y]+=a[x];
            del[y]-=a[x];
            sum+=a[x];
        }
        else{
            cin>>y;
            pre[y]+=del[y];
            sum+=del[y];
            del[y]=0;
        }
        if(sum==to) cout<<"YES"<<endl;
        else cout<<"NO"<<endl;
    }
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    int T=1;//cin>>T;
    while(T--) solve();
    return 0;
}
```
