从势能上分析题目，发现从一个点到另一个点的势能改变情况有多种时你就可以通过走反向边与环无限扩大代价。

### 方案一

然后做法就比较显然了，对于每一个图上的联通块，规定一个`0`势能点，然后建dfs生成树计算出每个点的势能，如果返祖边所指向的势能点是与计算的势能差不同的就说明可以无限扩大代价 ，回答`inf`。

如果是两个联通块中的点，那就永远无法抵达，回答`nan`。

否则就回答势能差值。
```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
void solve(){
    int n,m,q;
    cin>>n>>m>>q;
    vector<vector<pair<int,ll>>>G(n+1);
    for(int i=1;i<=m;i++){
        int x,y,w;cin>>x>>y>>w;
        G[x].emplace_back(y,w);
        G[y].emplace_back(x,-w);
    }
    vector<int>vis(n+1);
    int idx=0;
    vector<ll>d(n+1);
    vector<int>f(n+1,true);
    auto dfs=[&](auto &&dfs,int u) ->void {
        for(auto &[v,w]:G[u]){
            if(!vis[v]){
                vis[v]=idx;
                d[v]=d[u]+w;
                dfs(dfs,v);
            }
            else{
                if(d[v]!=d[u]+w) f[idx]=false;    
            }
        }
    };
    for(int i=1;i<=n;i++){
        if(!vis[i]){
            ++idx;
            vis[i]=idx;
            dfs(dfs,i);
        }
    }
    while(q--){
        int x,y;cin>>x>>y;
        if(vis[x]!=vis[y]) cout<<"nan"<<endl;
        else if(!f[vis[x]]) cout<<"inf"<<endl;
        else cout<<d[y]-d[x]<<endl;
    }
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    solve();
    return 0;
}
```
### 方案二

当然其实不用这么麻烦，维护的只是若干点对的距离差。可以建立若干个等式，然后用`带权并查集`检验是否成立。
```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

struct DSU {
    vector<int> f, bad;
    vector<ll> dis; // dis[x] = pot[x] - pot[f[x]]

    DSU(int n) {
        f.resize(n + 1);
        bad.assign(n + 1, 0);
        dis.assign(n + 1, 0);
        for (int i = 1; i <= n; i++) f[i] = i;
    }

    int find(int x) {
        if (f[x] == x) return x;
        int p = f[x];
        int r = find(p);
        dis[x] += dis[p]; // 更新为 pot[x] - pot[r]
        f[x] = r;
        return r;
    }

    // 添加约束 pot[B] - pot[A] = C
    void unite(int A, int B, ll C) {
        int ra = find(A), rb = find(B);
        if (ra == rb) {
            if (dis[B] - dis[A] != C) {
                bad[ra] = 1; // 矛盾，整个连通分量 inf
            }
            return;
        }
        // 把 rb 挂到 ra 下
        f[rb] = ra;
        dis[rb] = C + dis[A] - dis[B];
        bad[ra] |= bad[rb];
    }

    // 返回 0: 正常，1: nan，2: inf
    int query(int X, int Y, ll &ans) {
        int rx = find(X), ry = find(Y);
        if (rx != ry) return 1;
        if (bad[rx]) return 2;
        ans = dis[Y] - dis[X];
        return 0;
    }
};
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int N, M, Q;
    cin >> N >> M >> Q;
    DSU dsu(N);
    for (int i = 0; i < M; i++) {
        int A, B;
        ll C;
        cin >> A >> B >> C;
        dsu.unite(A, B, C);
    }
    while (Q--) {
        int X, Y;
        cin >> X >> Y;
        ll ans;
        int res = dsu.query(X, Y, ans);
        if (res == 1) {
            cout << "nan\n";
        } else if (res == 2) {
            cout << "inf\n";
        } else {
            cout << ans << '\n';
        }
    }
    return 0;
}
```
