观察到是典型的 Nim 游戏变种，答案一定是多堆 `sg` 值的异或和判定，先简单写个sg打表函数
```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const int N=1e6;
int dp[N+5];
int sg(int x){
    if(x<=0) return 0;
    if(dp[x]!=-1) return dp[x];
    int y=sqrt(x);
    vector<int>vis(y+1);
    for(int k=(y*y==x?1:0);k<=y;k++) {
        int cur=sg((y-k)*y);
        if(cur>y) continue;
        vis[cur]++;
    }
    int mex=0;
    for(int j=0;j<=y;j++){
        if(vis[j]) mex++; 
        else break;
    }
    return dp[x]=mex;
}
void solve(){
    int n;cin>>n;
    int f=0,x;
    for(int i=1;i<=n;i++){
        cin>>x;
        f^=sg(x);
    }
    if(f) cout<<"First"<<endl;
    else cout<<"Second"<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    memset(dp,-1,sizeof dp);
    int T=1;//cin>>T
    while(T--) solve();
    return 0;
}
```

然后发现跑了将近 1s,放在3年前的比赛大概率是超时的。思考如何优化，发现可以归并sg值后继状态相同的数，这样每个数只考虑 $\lfloor \sqrt{a_i} \rfloor$ 种状态，写一下代码跑了 `46ms`

```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'
typedef long long ll;
const int N=2e3;
int dp1[N+5],dp2[N+5];
int sg(int x){
    if(x<=0) return 0;
    int y=sqrt(x);
    if(y*y==x&&dp1[y]!=-1) return dp1[y];
    else if(y*y!=x&&dp2[y]!=-1) return dp2[y];
    vector<int>vis(y+1);
    for(int k=(y*y==x?1:0);k<=y;k++) {
        int cur=sg((y-k)*y);
        if(cur>y) continue;
        vis[cur]++;
    }
    int mex=0;
    for(int j=0;j<=y;j++){
        if(vis[j]) mex++; 
        else break;
    }
    if(y*y==x) return dp1[y]=mex;
    return dp2[y]=mex;
}
void solve(){
    int n;cin>>n;
    int f=0,x;
    for(int i=1;i<=n;i++){
        cin>>x;
        f^=sg(x);
    }
    if(f) cout<<"First"<<endl;
    else cout<<"Second"<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    memset(dp1,-1,sizeof dp1);
    memset(dp2,-1,sizeof dp2);
    int T=1;//cin>>T
    while(T--) solve();
    return 0;
}
```
