题目要求 $p_i+i=p_i \oplus i$，不妨思考 $a+b=a\oplus b$ 何时成立？
发现当且仅当 $ a\&b=0$式子成立。这样问题就转化为了构造排列与下标按位与为`0`。

一个有效的构造方式是从高位到低位弥补满位掩码
```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define endl '\n'
void solve(){
    int n;cin>>n;
    vector<ll>a(n);
    ll mx=0;
    for(int i=n-1;i>=0;i--){
        if(a[i]||i==0) continue;
        ll s=(1<<__lg(i)+1)-1;
        a[(s^i)]=i;
        a[i]=(s^i);
    }
    for(int i=0;i<n;i++) cout<<a[i]<<' ';
    cout<<endl;
}
int main(){
    cin.tie(0)->ios::sync_with_stdio(false);
    solve();
    return 0;
}
```
